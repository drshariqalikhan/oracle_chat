from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any


class LLMProviderError(Exception):
    def __init__(self, message: str, status_code: int = 502) -> None:
        super().__init__(message)
        self.message = message
        self.status_code = status_code


@dataclass(frozen=True)
class LLMResponse:
    content: str
    model: str


class LLMProvider(ABC):
    @abstractmethod
    async def chat(
        self,
        messages: list[dict[str, Any]],
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        """Send chat messages and return an LLMResponse containing text and resolved model."""
        raise NotImplementedError

    @abstractmethod
    async def analyze_image(
        self,
        prompt: str,
        image_data_url: str,
        history: list[dict[str, Any]] | None = None,
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        """Analyze an image using a multimodal model and return an LLMResponse."""
        raise NotImplementedError

    @abstractmethod
    async def process_document(
        self,
        prompt: str,
        document_context: str,
        history: list[dict[str, Any]] | None = None,
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        """Answer a prompt grounded in extracted document text."""
        raise NotImplementedError

from backend.config import Settings
from backend.llm.base import LLMProvider, LLMProviderError
from backend.llm.ollama import OllamaProvider
from backend.llm.openrouter import OpenRouterProvider

SUPPORTED_PROVIDERS = ("openrouter", "ollama")


def create_llm_provider(
    settings: Settings,
    provider_name: str | None = None,
    api_key: str | None = None,
    default_model: str | None = None,
    ollama_base_url: str | None = None,
) -> tuple[str, LLMProvider]:
    selected_provider = (provider_name or settings.llm_provider or "openrouter").strip().lower()

    if selected_provider == "openrouter":
        resolved_key = (api_key or "").strip() or settings.openrouter_api_key
        resolved_model = (default_model or "").strip() or settings.model_name or "openrouter/free"
        return selected_provider, OpenRouterProvider(
            api_key=resolved_key,
            default_model=resolved_model,
            base_url=settings.openrouter_base_url,
        )

    if selected_provider == "ollama":
        resolved_url = (ollama_base_url or "").strip().rstrip("/") or settings.ollama_base_url
        env_default = "" if settings.model_name == "openrouter/free" else settings.model_name
        resolved_model = (default_model or "").strip() or env_default
        return selected_provider, OllamaProvider(
            base_url=resolved_url,
            default_model=resolved_model,
            timeout=settings.ollama_timeout,
        )

    raise LLMProviderError(
        f"Unsupported LLM provider '{selected_provider}'. Supported providers: {', '.join(SUPPORTED_PROVIDERS)}.",
        status_code=400,
    )

=== FILE: backend/llm/ollama.py ===
import re
from typing import Any
from urllib.parse import urlparse

import httpx

from backend.llm.base import LLMProvider, LLMProviderError, LLMResponse

CLOUDSHELL_PREVIEW_RE = re.compile(
    r"^https?://(\d+)-cs-[^/]+\.cloudshell\.dev(?:/.*)?$",
    re.IGNORECASE,
)


def normalize_ollama_base_url(base_url: str) -> str:
    cleaned = (base_url or "").strip().rstrip("/")
    match = CLOUDSHELL_PREVIEW_RE.match(cleaned)
    if match:
        port = match.group(1)
        return f"http://127.0.0.1:{port}"
    return cleaned


class OllamaProvider(LLMProvider):
    def __init__(
        self,
        base_url: str = "http://localhost:11434",
        default_model: str = "",
        timeout: float = 120.0,
    ) -> None:
        self.base_url = normalize_ollama_base_url(base_url or "http://localhost:11434")
        self.default_model = default_model.strip()
        self.timeout = timeout

    def _validate_base_url(self) -> None:
        parsed = urlparse(self.base_url)
        if parsed.scheme not in ("http", "https") or not parsed.netloc:
            raise LLMProviderError(
                f"Invalid Ollama URL: '{self.base_url}'. Must start with http:// or https://",
                status_code=400,
            )

    async def list_models(self) -> list[dict[str, Any]]:
        self._validate_base_url()
        tags_url = f"{self.base_url}/api/tags"

        try:
            async with httpx.AsyncClient(timeout=8.0) as client:
                response = await client.get(tags_url, headers={"Accept": "application/json"})
        except httpx.TimeoutException as exc:
            raise LLMProviderError(
                f"Timed out connecting to Ollama at {self.base_url}",
                status_code=504,
            ) from exc
        except httpx.RequestError as exc:
            raise LLMProviderError(
                f"Cannot reach Ollama at {self.base_url} ({exc})",
                status_code=502,
            ) from exc

        if response.status_code >= 400:
            detail = self._extract_error_detail(response)
            raise LLMProviderError(
                f"Ollama returned HTTP {response.status_code} at {tags_url}: {detail}",
                status_code=502,
            )

        try:
            payload = response.json()
        except ValueError as exc:
            raise LLMProviderError(
                f"Invalid JSON received from Ollama at {tags_url}.",
                status_code=502,
            ) from exc

        raw_models = payload.get("models", [])
        return [
            {
                "name": item.get("name", ""),
                "size": item.get("size", 0),
                "modified_at": item.get("modified_at", ""),
                "details": item.get("details", {}),
            }
            for item in raw_models
            if item.get("name")
        ]

    async def chat(
        self,
        messages: list[dict[str, Any]],
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        self._validate_base_url()
        selected_model = (model or "").strip() or self.default_model
        if not selected_model or selected_model == "openrouter/free":
            raise LLMProviderError(
                "No Ollama model specified. Select a local model in Settings or set MODEL_NAME in .env.",
                status_code=400,
            )

        url = f"{self.base_url}/api/chat"
        payload: dict[str, Any] = {
            "model": selected_model,
            "messages": messages,
            "stream": False,
        }

        try:
            async with httpx.AsyncClient(timeout=self.timeout) as client:
                response = await client.post(url, json=payload)
        except httpx.TimeoutException as exc:
            raise LLMProviderError(
                f"Timed out waiting for Ollama response at {self.base_url} (model: {selected_model}).",
                status_code=504,
            ) from exc
        except httpx.RequestError as exc:
            raise LLMProviderError(
                f"Cannot reach Ollama at {self.base_url}: {exc}",
                status_code=502,
            ) from exc

        if response.status_code >= 400:
            detail = self._extract_error_detail(response)
            if response.status_code == 404:
                raise LLMProviderError(
                    f"Ollama model '{selected_model}' not found ({detail}). Run: ollama pull {selected_model}",
                    status_code=404,
                )
            mapped_status = 400 if response.status_code == 400 else 502
            raise LLMProviderError(
                f"Ollama error ({response.status_code}): {detail}",
                status_code=mapped_status,
            )

        try:
            data = response.json()
        except ValueError as exc:
            raise LLMProviderError(
                f"Invalid JSON response received from Ollama at {self.base_url}.",
                status_code=502,
            ) from exc

        message_obj = data.get("message") or {}
        content = message_obj.get("content")
        if not isinstance(content, str) or not content.strip():
            raise LLMProviderError(
                f"Ollama returned an empty response for model '{selected_model}'.",
                status_code=502,
            )

        resolved_model = str(data.get("model") or selected_model).strip()
        return LLMResponse(content=content.strip(), model=resolved_model)

    async def analyze_image(
        self,
        prompt: str,
        image_data_url: str,
        history: list[dict[str, Any]] | None = None,
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        raw_base64 = (
            image_data_url.split(",", 1)[1]
            if "," in image_data_url
            else image_data_url
        )
        multimodal_message = {
            "role": "user",
            "content": prompt,
            "images": [raw_base64],
        }
        messages = [*(history or []), multimodal_message]
        return await self.chat(messages=messages, model=model, **kwargs)

    async def process_document(
        self,
        prompt: str,
        document_context: str,
        history: list[dict[str, Any]] | None = None,
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        context_instruction = {
            "role": "user",
            "content": (
                "Use the following uploaded document content to answer the user's questions accurately. "
                "If the answer is not in the document, say so clearly.\n\n"
                f"{document_context}"
            ),
        }
        messages = [
            context_instruction,
            *(history or []),
            {"role": "user", "content": prompt},
        ]
        return await self.chat(messages=messages, model=model, **kwargs)

    @staticmethod
    def _extract_error_detail(response: httpx.Response) -> str:
        try:
            data = response.json()
            if isinstance(data, dict) and data.get("error"):
                return str(data["error"])
        except ValueError:
            pass
        return response.text.strip() or f"HTTP {response.status_code}"

from typing import Any

import httpx

from backend.llm.base import LLMProvider, LLMProviderError, LLMResponse


class OpenRouterProvider(LLMProvider):
    def __init__(
        self,
        api_key: str,
        default_model: str = "openrouter/free",
        base_url: str = "https://openrouter.ai/api/v1",
        timeout: float = 90.0,
    ) -> None:
        self.api_key = api_key.strip()
        self.default_model = default_model.strip()
        self.base_url = base_url.strip().rstrip("/")
        self.timeout = timeout

    async def chat(
        self,
        messages: list[dict[str, Any]],
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        if not self.api_key:
            raise LLMProviderError(
                "OpenRouter API key is missing. Set OPENROUTER_API_KEY in .env or enter it in Settings.",
                status_code=400,
            )

        selected_model = (model or "").strip() or self.default_model
        if not selected_model:
            raise LLMProviderError("No OpenRouter model specified.", status_code=400)

        url = f"{self.base_url}/chat/completions"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
            "HTTP-Referer": "https://github.com/drshariqalikhan/oracle_chat",
            "X-Title": "ORACLE CHAT",
        }
        payload: dict[str, Any] = {
            "model": selected_model,
            "messages": messages,
        }

        try:
            async with httpx.AsyncClient(timeout=self.timeout) as client:
                response = await client.post(url, headers=headers, json=payload)
        except httpx.TimeoutException as exc:
            raise LLMProviderError(
                f"Timed out waiting for OpenRouter response (model: {selected_model}).",
                status_code=504,
            ) from exc
        except httpx.RequestError as exc:
            raise LLMProviderError(
                f"Network error contacting OpenRouter: {exc}",
                status_code=502,
            ) from exc

        if response.status_code >= 400:
            detail = self._extract_error_detail(response)
            mapped_status = (
                response.status_code
                if response.status_code in (400, 401, 402, 403, 404, 429)
                else 502
            )
            raise LLMProviderError(
                f"OpenRouter error ({response.status_code}): {detail}",
                status_code=mapped_status,
            )

        try:
            data = response.json()
        except ValueError as exc:
            raise LLMProviderError("Invalid JSON response received from OpenRouter.", status_code=502) from exc

        choices = data.get("choices") or []
        if not choices:
            raise LLMProviderError("OpenRouter returned no choices in response.", status_code=502)

        message_obj = choices[0].get("message") or {}
        content = message_obj.get("content")

        if isinstance(content, list):
            text_parts = [
                part.get("text", "")
                for part in content
                if isinstance(part, dict) and part.get("type") == "text"
            ]
            content = "\n".join(part for part in text_parts if part)

        if not isinstance(content, str) or not content.strip():
            raise LLMProviderError("OpenRouter returned an empty message.", status_code=502)

        assigned_model = str(data.get("model") or selected_model).strip()
        return LLMResponse(content=content.strip(), model=assigned_model)

    async def analyze_image(
        self,
        prompt: str,
        image_data_url: str,
        history: list[dict[str, Any]] | None = None,
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        multimodal_message = {
            "role": "user",
            "content": [
                {"type": "text", "text": prompt},
                {"type": "image_url", "image_url": {"url": image_data_url}},
            ],
        }
        messages = [*(history or []), multimodal_message]
        return await self.chat(messages=messages, model=model, **kwargs)

    async def process_document(
        self,
        prompt: str,
        document_context: str,
        history: list[dict[str, Any]] | None = None,
        model: str | None = None,
        **kwargs: Any,
    ) -> LLMResponse:
        context_instruction = {
            "role": "user",
            "content": (
                "Use the following uploaded document content to answer the user's questions accurately. "
                "If the answer is not in the document, say so clearly.\n\n"
                f"{document_context}"
            ),
        }
        messages = [
            context_instruction,
            *(history or []),
            {"role": "user", "content": prompt},
        ]
        return await self.chat(messages=messages, model=model, **kwargs)

    @staticmethod
    def _extract_error_detail(response: httpx.Response) -> str:
        try:
            data = response.json()
            if isinstance(data, dict):
                err = data.get("error")
                if isinstance(err, dict) and err.get("message"):
                    return str(err["message"])
                if data.get("message"):
                    return str(data["message"])
        except ValueError:
            pass
        return response.text.strip() or f"HTTP {response.status_code}"

async def _parse_with_optional_repair(
    provider: LLMProvider,
    initial_result: LLMResponse,
    base_history: list[dict[str, Any]],
    user_prompt: str,
    ui_mode: str,
) -> tuple[StandardizedLLMOutput, LLMResponse]:
    standardized = a2ui_service.parse_and_standardize(
        initial_result.content,
        ui_mode=ui_mode,
    )

    if ui_mode == "text" or standardized.used_a2ui or not standardized.validation_warnings:
        return standardized, initial_result

    repair_messages = [
        *base_history,
        {"role": "user", "content": user_prompt},
        {"role": "assistant", "content": initial_result.content},
        {"role": "user", "content": a2ui_service.build_repair_prompt(standardized.validation_warnings)},
    ]

    try:
        repaired_result = await provider.chat(
            messages=repair_messages,
            model=initial_result.model,
        )
    except LLMProviderError:
        return standardized, initial_result

    repaired_standardized = a2ui_service.parse_and_standardize(
        repaired_result.content,
        ui_mode=ui_mode,
    )
    repaired_standardized = repaired_standardized.model_copy(update={"repair_retried": True})

    if repaired_standardized.used_a2ui:
        return repaired_standardized, repaired_result

    combined_warnings = [
        *standardized.validation_warnings,
        *repaired_standardized.validation_warnings,
    ]
    return (
        repaired_standardized.model_copy(update={"validation_warnings": combined_warnings}),
        repaired_result,
    )


@router.post("/chat", response_model=ChatResponse)
async def chat(payload: ChatRequest) -> ChatResponse:
    global _pinned_free_model, _pinned_vision_model

    settings = get_settings()
    session_id = (payload.session_id or "").strip() or memory_service.create_session_id()

    try:
        active_provider, provider = create_llm_provider(
            settings=settings,
            provider_name=payload.provider,
            api_key=payload.api_key,
            default_model=payload.model,
            ollama_base_url=payload.ollama_base_url,
        )
    except LLMProviderError as exc:
        raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc

    if active_provider == "openrouter":
        requested_model = (payload.model or "").strip() or settings.model_name or FREE_ROUTER_MODEL
    else:
        env_ollama_model = "" if settings.model_name == FREE_ROUTER_MODEL else settings.model_name
        requested_model = (payload.model or "").strip() or env_ollama_model

    is_vision = bool(payload.image_id)
    image_data_url: str | None = None

    if payload.image_id:
        try:
            image_data_url = image_service.load_image_as_data_url(payload.image_id)
        except ImageValidationError as exc:
            raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc

    if payload.document_id:
        try:
            doc_text, _, _ = file_service.extract_text_by_id(payload.document_id)
        except FileValidationError as exc:
            raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc

        doc_display_name = payload.document_name or payload.document_id
        memory_service.attach_document(
            session_id=session_id,
            file_id=payload.document_id,
            filename=doc_display_name,
            text=doc_text,
        )

    prior_messages = memory_service.get_history(session_id)
    documents_context = memory_service.get_documents_context(session_id)
    has_document_context = bool(documents_context)
    a2ui_system_prompt = a2ui_service.build_system_prompt(payload.ui_mode)

    base_history: list[dict[str, Any]] = []
    if a2ui_system_prompt:
        base_history.append({"role": "system", "content": a2ui_system_prompt})
    base_history.extend(prior_messages)

    async def run_inference(target_model: str | None) -> LLMResponse:
        if is_vision and image_data_url is not None:
            vision_history = list(base_history)
            if documents_context:
                vision_history.append(
                    {
                        "role": "user",
                        "content": f"Uploaded document context:\n\n{documents_context}",
                    }
                )
            return await provider.analyze_image(
                prompt=payload.message,
                image_data_url=image_data_url,
                history=vision_history,
                model=target_model,
            )

        if documents_context:
            return await provider.process_document(
                prompt=payload.message,
                document_context=documents_context,
                history=base_history,
                model=target_model,
            )

        return await provider.chat(
            messages=[*base_history, {"role": "user", "content": payload.message}],
            model=target_model,
        )

    memory_prefixes: list[str] = []
    if payload.image_id:
        memory_prefixes.append(f"[Uploaded image: {payload.image_id}]")
    if payload.document_id:
        memory_prefixes.append(
            f"[Uploaded document: {payload.document_name or payload.document_id}]"
        )
    memory_user_text = (
        f"{' '.join(memory_prefixes)} {payload.message}"
        if memory_prefixes
        else payload.message
    )

    if active_provider == "openrouter" and requested_model == FREE_ROUTER_MODEL:
        if is_vision:
            active_pinned = (payload.pinned_vision_model or "").strip() or _pinned_vision_model
        else:
            active_pinned = (payload.pinned_model or "").strip() or _pinned_free_model

        target_model = active_pinned or FREE_ROUTER_MODEL

        try:
            result = await run_inference(target_model)
        except LLMProviderError as exc:
            if target_model != FREE_ROUTER_MODEL and exc.status_code in RETRIABLE_PIN_STATUSES:
                if is_vision:
                    _pinned_vision_model = None
                else:
                    _pinned_free_model = None
                try:
                    result = await run_inference(FREE_ROUTER_MODEL)
                except LLMProviderError as retry_exc:
                    raise HTTPException(
                        status_code=retry_exc.status_code,
                        detail=retry_exc.message,
                    ) from retry_exc
            else:
                raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc

        if result.model and result.model != FREE_ROUTER_MODEL:
            if is_vision:
                _pinned_vision_model = result.model
            else:
                _pinned_free_model = result.model

        standardized, final_result = await _parse_with_optional_repair(
            provider=provider,
            initial_result=result,
            base_history=base_history,
            user_prompt=payload.message,
            ui_mode=payload.ui_mode,
        )

        updated_history = memory_service.append_turn(
            session_id=session_id,
            user_message=memory_user_text,
            assistant_reply=standardized.text,
        )

        return ChatResponse(
            reply=standardized.text,
            output=standardized,
            session_id=session_id,
            history_length=len(updated_history),
            provider=active_provider,
            model=final_result.model,
            pinned=bool(_pinned_vision_model if is_vision else _pinned_free_model),
            vision_used=is_vision,
            document_used=has_document_context,
        )

    try:
        result = await run_inference(requested_model or None)
    except LLMProviderError as exc:
        raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc

    standardized, final_result = await _parse_with_optional_repair(
        provider=provider,
        initial_result=result,
        base_history=base_history,
        user_prompt=payload.message,
        ui_mode=payload.ui_mode,
    )

    updated_history = memory_service.append_turn(
        session_id=session_id,
        user_message=memory_user_text,
        assistant_reply=standardized.text,
    )

    return ChatResponse(
        reply=standardized.text,
        output=standardized,
        session_id=session_id,
        history_length=len(updated_history),
        provider=active_provider,
        model=final_result.model,
        pinned=False,
        vision_used=is_vision,
        document_used=has_document_context,
    )

import re
from typing import Any

from a2ui.core.basic_catalog import BasicCatalog
from a2ui.core.schema import ProtocolVersion
from a2ui.inference_formats import ExpressFormatFactory
from a2ui.parser import A2uiPart, TextPart, to_message_dicts
from a2ui.processor import A2uiGenerator, CatalogConfig
from a2ui.utils import validate_payload

from backend.models import StandardizedLLMOutput

A2UI_BLOCK_PATTERN = re.compile(r"<a2ui>.*?</a2ui>", re.DOTALL | re.IGNORECASE)
ALL_BASIC_COMPONENTS = (
    "Row",
    "Column",
    "List",
    "Card",
    "Tabs",
    "Modal",
    "Divider",
    "Text",
    "Image",
    "Icon",
    "Video",
    "AudioPlayer",
    "Button",
    "TextField",
    "CheckBox",
    "ChoicePicker",
    "Slider",
    "DateTimeInput",
)

STRICT_EXPRESS_RULES = """STRICT A2UI EXPRESS RULES:
1. Always start inside <a2ui>...</a2ui> with: root = Card(...) or root = Column(...)
2. NEVER invent custom surface names (e.g. do NOT write "demo" or createSurface("name")).
3. NEVER use slash commands (e.g. /updateComponents) or raw JSON inside <a2ui>...</a2ui>.
4. All string literals must be quoted ("..."), and Text.variant may only be "body" or "caption".
5. Always provide a concise plain-text reply outside <a2ui>...</a2ui> tags."""

MINIMAL_EXPRESS_EXAMPLE = """Minimal valid example:
Here is a quick option card.
<a2ui>
root = Card(Column([
  Text("Quick Summary", variant="body"),
  Button(Text("Confirm"), variant="primary", action=Event("confirm_action"))
]))
</a2ui>"""


class A2uiService:
    def __init__(self) -> None:
        self.catalog = BasicCatalog(ProtocolVersion.V1_0)
        self.generator = A2uiGenerator(
            [CatalogConfig(self.catalog)],
            inference_format_factory=ExpressFormatFactory(),
        )
        self.capabilities = {
            "1.0": {
                "supportedCatalogIds": [self.catalog.catalog_id],
            }
        }
        self._base_processor = self.generator.create_processor(self.capabilities)

    def build_system_prompt(self, ui_mode: str) -> str | None:
        if ui_mode == "text":
            return None

        if ui_mode == "a2ui":
            mode_rule = (
                "ALWAYS include a valid <a2ui>...</a2ui> Express UI block along with a brief 1-2 sentence "
                "plain-text summary outside the tags.\n\n"
                f"{MINIMAL_EXPRESS_EXAMPLE}"
            )
        else:
            mode_rule = (
                "Be CONSERVATIVE with <a2ui>: reply in plain text ONLY for normal conversation, Q&A, "
                "explanations, and document summaries. ONLY include <a2ui>...</a2ui> when the user "
                "explicitly asks for a form, interactive card, checklist, choices, slider, or UI widget."
            )

        return (
            "You are ORACLE CHAT, a helpful assistant capable of generating interactive A2UI v1.0 interfaces.\n"
            f"{mode_rule}\n\n"
            f"{STRICT_EXPRESS_RULES}\n\n"
            f"{self._base_processor.prompt_snippet}"
        )

    @staticmethod
    def build_repair_prompt(warnings: list[str]) -> str:
        error_summary = "; ".join(warnings)
        return (
            "Your previous <a2ui>...</a2ui> block failed A2UI Express validation with:\n"
            f"{error_summary}\n\n"
            "Rewrite your response with a valid <a2ui>...</a2ui> block following these exact rules:\n"
            "- Start with: root = Card(...) or root = Column(...)\n"
            "- Do NOT invent custom surface names or slash commands.\n"
            "- Text.variant can only be \"body\" or \"caption\".\n"
            "- Keep a brief plain-text summary outside <a2ui>...</a2ui>."
        )

    def parse_and_standardize(self, raw_reply: str, ui_mode: str = "auto") -> StandardizedLLMOutput:
        cleaned_raw = (raw_reply or "").strip()
        if ui_mode == "text":
            return StandardizedLLMOutput(
                text=cleaned_raw,
                a2ui_messages=[],
                used_a2ui=False,
                ui_mode="text",
                raw_format="text",
            )

        processor = self.generator.create_processor(self.capabilities)
        warnings: list[str] = []

        try:
            parts = processor.parse_response(cleaned_raw)

            fallback_text = A2UI_BLOCK_PATTERN.sub("", cleaned_raw).strip() or cleaned_raw
            return StandardizedLLMOutput(
                text=fallback_text,
                a2ui_messages=[],
                used_a2ui=False,

                raw_format="text",
                validation_warnings=warnings,

        text_segments: list[str] = []
        wire_messages: list[dict[str, Any]] = []

        for part in parts:
            if isinstance(part, TextPart):
                if part.text and part.text.strip():
                    text_segments.append(part.text.strip())
            elif isinstance(part, A2uiPart):
                wire_messages.extend(to_message_dicts(part.a2ui))

        if wire_messages:
            try:
                validate_payload([self.catalog], wire_messages)
            except Exception as exc:
                warnings.append(f"A2UI catalog validation fallback: {exc}")
                fallback_text = (
                    "\n\n".join(text_segments).strip()
                    or A2UI_BLOCK_PATTERN.sub("", cleaned_raw).strip()
                    or cleaned_raw
                )
                return StandardizedLLMOutput(
                    text=fallback_text,
                    a2ui_messages=[],
                    used_a2ui=False,
                    ui_mode=ui_mode,  # type: ignore[arg-type]
                    raw_format="text",
                    validation_warnings=warnings,
                )

        final_text = "\n\n".join(text_segments).strip()
        if not final_text:
            final_text = "Interactive A2UI surface rendered below." if wire_messages else cleaned_raw

        return StandardizedLLMOutput(
            text=final_text,
            a2ui_messages=wire_messages,
            used_a2ui=bool(wire_messages),

from dataclasses import dataclass
from pathlib import Path
from uuid import uuid4

from fastapi import UploadFile
from pypdf import PdfReader

ALLOWED_DOC_EXTENSIONS = {".pdf", ".txt"}
MAX_DOCUMENT_BYTES = 15 * 1024 * 1024  # 15 MB
MAX_EXTRACTED_CHARS = 20000


class FileValidationError(Exception):
    def __init__(self, message: str, status_code: int = 400) -> None:
        super().__init__(message)
        self.message = message
        self.status_code = status_code


@dataclass(frozen=True)
class SavedDocumentInfo:
    file_id: str
    original_filename: str
    file_type: str
    size_bytes: int
    page_count: int
    char_count: int
    truncated: bool
    extracted_text: str


class FileService:
    def __init__(self, uploads_dir: Path) -> None:
        self.uploads_dir = uploads_dir
        self.uploads_dir.mkdir(parents=True, exist_ok=True)

    async def save_document(self, file: UploadFile) -> SavedDocumentInfo:
        original_name = (file.filename or "document").strip()
        suffix = Path(original_name).suffix.lower()

        if suffix not in ALLOWED_DOC_EXTENSIONS:
            allowed_list = ", ".join(sorted(ALLOWED_DOC_EXTENSIONS))
            raise FileValidationError(
                f"Unsupported document extension '{suffix or 'none'}'. Allowed: {allowed_list}",
                status_code=400,
            )

        content = await file.read()
        if not content:
            raise FileValidationError("Uploaded document file is empty.", status_code=400)

        if len(content) > MAX_DOCUMENT_BYTES:
            raise FileValidationError(
                f"Document exceeds maximum allowed size of {MAX_DOCUMENT_BYTES // (1024 * 1024)} MB.",
                status_code=413,
            )

        saved_filename = f"{uuid4().hex}{suffix}"
        destination = self.uploads_dir / saved_filename
        destination.write_bytes(content)

        try:
            extracted_text, page_count = self._extract_from_path(destination, suffix)
        except FileValidationError:
            destination.unlink(missing_ok=True)
            raise

        truncated = False
        if len(extracted_text) > MAX_EXTRACTED_CHARS:
            extracted_text = (
                extracted_text[:MAX_EXTRACTED_CHARS].rstrip()
                + f"\n\n[Note: Document text truncated at {MAX_EXTRACTED_CHARS:,} characters.]"
            )
            truncated = True

        return SavedDocumentInfo(
            file_id=saved_filename,
            original_filename=original_name,
            file_type=suffix.lstrip("."),
            size_bytes=len(content),
            page_count=page_count,
            char_count=len(extracted_text),
            truncated=truncated,
            extracted_text=extracted_text,
        )

    def extract_text_by_id(self, file_id: str) -> tuple[str, int, bool]:
        cleaned_id = file_id.strip()
        if not cleaned_id or Path(cleaned_id).name != cleaned_id:
            raise FileValidationError("Invalid document file_id.", status_code=400)

        doc_path = self.uploads_dir / cleaned_id
        if not doc_path.is_file():
            raise FileValidationError(
                f"Uploaded document '{cleaned_id}' was not found.",
                status_code=404,
            )

        suffix = doc_path.suffix.lower()
        if suffix not in ALLOWED_DOC_EXTENSIONS:
            raise FileValidationError(
                f"Unsupported document extension '{suffix}'.",
                status_code=400,
            )

        text, page_count = self._extract_from_path(doc_path, suffix)
        truncated = False
        if len(text) > MAX_EXTRACTED_CHARS:
            text = (
                text[:MAX_EXTRACTED_CHARS].rstrip()
                + f"\n\n[Note: Document text truncated at {MAX_EXTRACTED_CHARS:,} characters.]"
            )
            truncated = True

        return text, page_count, truncated

    @staticmethod
    def _extract_from_path(file_path: Path, suffix: str) -> tuple[str, int]:
        if suffix == ".txt":
            raw_bytes = file_path.read_bytes()
            text = ""
            for encoding in ("utf-8", "utf-8-sig", "latin-1"):
                try:
                    text = raw_bytes.decode(encoding)
                    break
                except UnicodeDecodeError:
                    continue
            cleaned = text.strip()
            if not cleaned:
                raise FileValidationError("Text file contains no readable text.", status_code=400)
            return cleaned, 1

        try:
            reader = PdfReader(str(file_path))
        except Exception as exc:
            raise FileValidationError(
                f"Failed to read PDF file: {exc}",
                status_code=400,
            ) from exc

        pages_text: list[str] = []
        for idx, page in enumerate(reader.pages, start=1):
            page_content = (page.extract_text() or "").strip()
            if page_content:
                pages_text.append(f"[Page {idx}]\n{page_content}")

        combined = "\n\n".join(pages_text).strip()
        if not combined:
            raise FileValidationError(
                "No extractable text found in this PDF (it may be a scanned/image-only PDF).",
                status_code=400,
            )

        return combined, len(reader.pages)

import base64
import io
from dataclasses import dataclass
from pathlib import Path
from uuid import uuid4

from fastapi import UploadFile
from PIL import Image, UnidentifiedImageError

ALLOWED_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp"}
ALLOWED_PIL_FORMATS = {"JPEG", "PNG", "WEBP"}
EXTENSION_TO_MIME = {
    ".jpg": "image/jpeg",
    ".jpeg": "image/jpeg",
    ".png": "image/png",
    ".webp": "image/webp",
}
MAX_IMAGE_BYTES = 10 * 1024 * 1024  # 10 MB


class ImageValidationError(Exception):
    def __init__(self, message: str, status_code: int = 400) -> None:
        super().__init__(message)
        self.message = message
        self.status_code = status_code


@dataclass(frozen=True)
class SavedImageInfo:
    file_id: str
    original_filename: str
    url: str
    content_type: str
    size_bytes: int
    width: int
    height: int


class ImageService:
    def __init__(self, uploads_dir: Path) -> None:
        self.uploads_dir = uploads_dir
        self.uploads_dir.mkdir(parents=True, exist_ok=True)

    async def save_image(self, file: UploadFile) -> SavedImageInfo:
        original_name = (file.filename or "image").strip()
        suffix = Path(original_name).suffix.lower()

        if suffix not in ALLOWED_EXTENSIONS:
            allowed_list = ", ".join(sorted(ALLOWED_EXTENSIONS))
            raise ImageValidationError(
                f"Unsupported image extension '{suffix or 'none'}'. Allowed: {allowed_list}",
                status_code=400,
            )

        content = await file.read()
        if not content:
            raise ImageValidationError("Uploaded image file is empty.", status_code=400)

        if len(content) > MAX_IMAGE_BYTES:
            raise ImageValidationError(
                f"Image exceeds maximum allowed size of {MAX_IMAGE_BYTES // (1024 * 1024)} MB.",
                status_code=413,
            )

        try:
            with Image.open(io.BytesIO(content)) as img:
                img.verify()
            with Image.open(io.BytesIO(content)) as img:
                width, height = img.size
                pil_format = (img.format or "").upper()
        except (UnidentifiedImageError, OSError, ValueError) as exc:
            raise ImageValidationError(
                "Invalid or corrupted image file. Please upload a valid JPG, PNG, or WEBP image.",
                status_code=400,
            ) from exc

        if pil_format not in ALLOWED_PIL_FORMATS:
            raise ImageValidationError(
                f"Unsupported image format '{pil_format}'. Allowed: JPEG, PNG, WEBP.",
                status_code=400,
            )

        normalized_ext = ".jpg" if pil_format == "JPEG" else f".{pil_format.lower()}"
        saved_filename = f"{uuid4().hex}{normalized_ext}"
        destination = self.uploads_dir / saved_filename
        destination.write_bytes(content)

        content_type = file.content_type or EXTENSION_TO_MIME[normalized_ext]
        return SavedImageInfo(
            file_id=saved_filename,
            original_filename=original_name,
            url=f"/uploads/{saved_filename}",
            content_type=content_type,
            size_bytes=len(content),
            width=width,
            height=height,
        )

    def load_image_as_data_url(self, file_id: str) -> str:
        cleaned_id = file_id.strip()
        if not cleaned_id or Path(cleaned_id).name != cleaned_id:
            raise ImageValidationError("Invalid image file_id.", status_code=400)

        image_path = self.uploads_dir / cleaned_id
        if not image_path.is_file():
            raise ImageValidationError(
                f"Uploaded image '{cleaned_id}' was not found.",
                status_code=404,
            )

        suffix = image_path.suffix.lower()
        mime_type = EXTENSION_TO_MIME.get(suffix)
        if not mime_type:
            raise ImageValidationError(
                f"Unsupported image extension '{suffix}'.",
                status_code=400,
            )

        encoded = base64.b64encode(image_path.read_bytes()).decode("ascii")
        return f"data:{mime_type};base64,{encoded}"

from uuid import uuid4


class MemoryService:
    def __init__(self, max_messages: int = 40, max_doc_chars: int = 24000) -> None:
        self.max_messages = max_messages
        self.max_doc_chars = max_doc_chars
        self._sessions: dict[str, list[dict[str, str]]] = {}
        self._documents: dict[str, list[dict[str, str]]] = {}

    @staticmethod
    def create_session_id() -> str:
        return uuid4().hex

    def get_history(self, session_id: str) -> list[dict[str, str]]:
        return [dict(msg) for msg in self._sessions.get(session_id, [])]

    def append_turn(
        self,
        session_id: str,
        user_message: str,
        assistant_reply: str,
    ) -> list[dict[str, str]]:
        history = self._sessions.setdefault(session_id, [])
        history.append({"role": "user", "content": user_message})
        history.append({"role": "assistant", "content": assistant_reply})

        if len(history) > self.max_messages:
            self._sessions[session_id] = history[-self.max_messages :]

        return self.get_history(session_id)

    def attach_document(
        self,
        session_id: str,
        file_id: str,
        filename: str,
        text: str,
    ) -> None:
        docs = self._documents.setdefault(session_id, [])
        docs = [doc for doc in docs if doc["file_id"] != file_id]
        docs.append(
            {
                "file_id": file_id,
                "filename": filename,
                "text": text[: self.max_doc_chars],
            }
        )
        # Keep the most recent 2 documents in active context
        self._documents[session_id] = docs[-2:]

    def get_documents_context(self, session_id: str) -> str:
        docs = self._documents.get(session_id, [])
        if not docs:
            return ""
        sections = [
            f"Document: {doc['filename']}\n--- BEGIN DOCUMENT ---\n{doc['text']}\n--- END DOCUMENT ---"
            for doc in docs
        ]
        return "\n\n".join(sections)

    def clear_session(self, session_id: str) -> None:
        self._sessions.pop(session_id, None)
        self._documents.pop(session_id, None)

    python3 -m venv "${APP_DIR}/venv"
fi
"${APP_DIR}/venv/bin/pip" install --upgrade pip
"${APP_DIR}/venv/bin/pip" install -r "${APP_DIR}/requirements.txt"

echo "[3/6] Ensuring .env configuration exists..."
if [ ! -f "${APP_DIR}/.env" ]; then
    cp "${APP_DIR}/.env.example" "${APP_DIR}/.env"
    echo "Created .env from .env.example — remember to add your OPENROUTER_API_KEY or set LLM_PROVIDER=ollama."
fi

echo "[4/6] Installing and enabling Ollama..."
if ! command -v ollama >/dev/null 2>&1; then
    curl -fsSL https://ollama.com/install.sh | sh
fi
sudo systemctl enable ollama
sudo systemctl restart ollama

if [ "${PULL_MODELS}" = "true" ]; then
    echo "Pulling gemma3:4b (this may take a few minutes)..."
    ollama pull gemma3:4b
fi

echo "[5/6] Configuring OS firewall (iptables + ufw) for SSH (22) and App (${APP_PORT})..."
sudo ufw allow OpenSSH >/dev/null 2>&1 || true
sudo ufw allow "${APP_PORT}/tcp" >/dev/null 2>&1 || true
sudo iptables -C INPUT -p tcp --dport "${APP_PORT}" -j ACCEPT 2>/dev/null || \
    sudo iptables -I INPUT 6 -m state --state NEW -p tcp --dport "${APP_PORT}" -j ACCEPT
sudo netfilter-persistent save >/dev/null 2>&1 || true

echo "[6/6] Installing and starting systemd service (oracle-chat.service)..."
TMP_UNIT="/tmp/oracle-chat.service"
sed \
    -e "s|__APP_USER__|${APP_USER}|g" \
    -e "s|__APP_DIR__|${APP_DIR}|g" \
    -e "s|__APP_PORT__|${APP_PORT}|g" \
    "${APP_DIR}/deploy/oracle-chat.service" > "${TMP_UNIT}"

sudo cp "${TMP_UNIT}" /etc/systemd/system/oracle-chat.service
rm -f "${TMP_UNIT}"
sudo systemctl daemon-reload
sudo systemctl enable oracle-chat
sudo systemctl restart oracle-chat

echo "=== Setup Complete ==="
echo "Check service status: sudo systemctl status oracle-chat"
echo "Follow live logs:     sudo journalctl -u oracle-chat -f"
echo "Test locally:         curl -s http://localhost:${APP_PORT}/health"

    function renderA2uiMessages(messages, onAction) {
        const container = document.createElement("div");
        container.className = "a2ui-surfaces-container";
        const surfaces = new Map();

        const getOrCreateSurface = (surfaceId) => {
            if (!surfaces.has(surfaceId)) {
                surfaces.set(surfaceId, {
                    surfaceId,
                    rootId: "root",
                    components: new Map(),
                    dataModel: {},
                });
            }
            return surfaces.get(surfaceId);
        };

        for (const msg of messages || []) {
            if (!msg || typeof msg !== "object") continue;

            if (msg.createSurface) {
                const cs = msg.createSurface;
                const s = getOrCreateSurface(cs.surfaceId || "main");
                if (cs.root) s.rootId = cs.root;
                if (cs.dataModel && typeof cs.dataModel === "object") {
                    Object.assign(s.dataModel, cs.dataModel);
                }
                if (Array.isArray(cs.components)) {
                    for (const comp of cs.components) {
                        if (comp && comp.id) s.components.set(comp.id, comp);
                    }
                }
            }

            if (msg.updateDataModel) {
                const udm = msg.updateDataModel;
                const s = getOrCreateSurface(udm.surfaceId || "main");
                const val = "value" in udm ? udm.value : udm.contents;
                setByPointer(s.dataModel, udm.path || "/", val);
            }

            if (msg.updateComponents) {
                const uc = msg.updateComponents;
                const s = getOrCreateSurface(uc.surfaceId || "main");
                const comps = Array.isArray(uc.components) ? uc.components : [];
                for (const comp of comps) {
                    if (comp && comp.id) s.components.set(comp.id, comp);
                }
            }

            if (msg.deleteSurface && msg.deleteSurface.surfaceId) {
                surfaces.delete(msg.deleteSurface.surfaceId);
            }
        }

        for (const surface of surfaces.values()) {
            const surfaceEl = document.createElement("div");
            surfaceEl.className = "a2ui-surface";
            const rootId = surface.components.has(surface.rootId)
                ? surface.rootId
                : surface.components.keys().next().value;
            if (rootId) {
                surfaceEl.appendChild(renderComponent(surface, rootId, onAction));
            }
            container.appendChild(surfaceEl);
        }

        return container;
    }

    window.OracleA2uiRenderer = {
        renderA2uiMessages,
    };

function formatTimestamp(date = new Date()) {
    return date.toTimeString().split(" ")[0];
}

function addLog(level, source, message, details = null) {
    const entry = {
        timestamp: formatTimestamp(),
        level,
        source,
        message,
        details: details ? (typeof details === "string" ? details : JSON.stringify(details)) : "",
    };
    logEntries.push(entry);
    renderLogs();
    updateErrorBadge();
}

function updateErrorBadge() {
    const errorCount = logEntries.filter((entry) => entry.level === "ERROR").length;
    if (errorCount > 0) {
        logErrorBadge.textContent = String(errorCount);
        logErrorBadge.classList.remove("hidden");
    } else {
        logErrorBadge.classList.add("hidden");
    }
}

function formatLogLine(entry) {
    const suffix = entry.details ? ` | ${entry.details}` : "";
    return `[${entry.timestamp}] [${entry.level}] [${entry.source}] ${entry.message}${suffix}`;
}

function renderLogs() {
    const filtered = logEntries.filter((entry) => {
        if (activeLogFilter === "error") {
            return entry.level === "ERROR";
        }
        if (activeLogFilter === "event") {
            return entry.level !== "ERROR";
        }
        return true;
    });

    logOutput.innerHTML = "";
    if (filtered.length === 0) {
        const empty = document.createElement("div");
        empty.className = "log-entry level-info";
        empty.textContent = "No matching log entries yet.";
        logOutput.appendChild(empty);
        return;
    }

    for (const entry of filtered) {
        const line = document.createElement("div");
        line.className = `log-entry level-${entry.level.toLowerCase()}`;
        line.textContent = formatLogLine(entry);
        logOutput.appendChild(line);
    }

    logOutput.scrollTop = logOutput.scrollHeight;
}

function toggleLogDrawer(forceOpen = null) {
    const shouldOpen = forceOpen !== null ? forceOpen : logDrawer.classList.contains("hidden");
    logDrawer.classList.toggle("hidden", !shouldOpen);
    if (shouldOpen) {
        renderLogs();
    }
}

async function copyLogsToClipboard() {
    const text = logEntries.map(formatLogLine).join("\n") || "No log entries.";
    try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
            await navigator.clipboard.writeText(text);
        } else {
            const temp = document.createElement("textarea");
            temp.value = text;
            document.body.appendChild(temp);
            temp.select();
            document.execCommand("copy");
            temp.remove();
        }
        const original = copyLogsBtn.textContent;
        copyLogsBtn.textContent = "Copied!";
        addLog("INFO", "LOGS", `Copied ${logEntries.length} log entries to clipboard.`);
        setTimeout(() => {
            copyLogsBtn.textContent = original;
        }, 1500);
    } catch (err) {
        addLog("ERROR", "LOGS", "Failed to copy logs to clipboard.", err.message);
    }
}

function generateClientSessionId() {
    if (window.crypto && typeof window.crypto.randomUUID === "function") {
        return window.crypto.randomUUID().replace(/-/g, "");
    }
    return `sess_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
}

function loadOrCreateSessionId() {
    try {
        const existing = localStorage.getItem(SESSION_STORAGE_KEY);
        if (existing && existing.trim()) {
            return existing.trim();
        }
        const created = generateClientSessionId();
        localStorage.setItem(SESSION_STORAGE_KEY, created);
        return created;
    } catch (err) {
        return generateClientSessionId();
    }
}

async function startNewChat() {
    if (isSending || isUploading) {
        return;
    }

    const previousSessionId = currentSessionId;
    currentSessionId = generateClientSessionId();
    try {
        localStorage.setItem(SESSION_STORAGE_KEY, currentSessionId);
    } catch (err) {
        // Ignore storage errors
    }

    clearPendingAttachment();
    chatHistory.innerHTML = "";
    chatHistory.appendChild(emptyStateTemplate.cloneNode(true));
    closeSettingsPage();
    messageInput.focus();

    try {
        await fetch(`/chat/history/${encodeURIComponent(previousSessionId)}`, {
            method: "DELETE",
        });
    } catch (err) {
        // Ignore cleanup errors
    }

    addLog("INFO", "SESSION", "Started new chat session.", {
        previousSessionId,
        newSessionId: currentSessionId,
    });
}

async function restoreSessionHistory() {
    try {
        const response = await fetch(`/chat/history/${encodeURIComponent(currentSessionId)}`);
        if (!response.ok) {
            return;
        }
        const data = await response.json();
        const messages = data.messages || [];
        if (messages.length > 0) {
            chatHistory.innerHTML = "";
            for (const msg of messages) {
                if (msg.role === "user" || msg.role === "assistant") {
                    appendMessage(msg.role, msg.content || "");
                }
            }
            addLog("INFO", "SESSION", "Restored session conversation history.", {
                sessionId: currentSessionId,
                messagesCount: messages.length,
            });
        }
    } catch (err) {
        addLog("WARN", "SESSION", "Could not restore session history.", err.message);
    }
}

function loadSettings() {
    try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (!raw) {
            return { ...DEFAULT_SETTINGS };
        }
        const parsed = JSON.parse(raw);
        if (parsed.openrouterModel === "google/gemma-4-31b-it:free") {
            parsed.openrouterModel = DEFAULT_SETTINGS.openrouterModel;
        }
        if (typeof parsed.ollamaBaseUrl === "string" && parsed.ollamaBaseUrl.includes(".cloudshell.dev")) {
            parsed.ollamaBaseUrl = "";
        }
        return { ...DEFAULT_SETTINGS, ...parsed };
    } catch (err) {
        return { ...DEFAULT_SETTINGS };
    }
}

function saveSettingsToStorage(settings) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
}

function formatBytes(bytes) {
    if (!bytes || Number.isNaN(bytes)) {
        return "";
    }
    const gb = bytes / (1024 ** 3);
    if (gb >= 1) {
        return `${gb.toFixed(1)} GB`;
    }
    const mb = bytes / (1024 ** 2);
    if (mb >= 1) {
        return `${mb.toFixed(1)} MB`;
    }
    return `${Math.max(1, Math.round(bytes / 1024))} KB`;
}

async function discoverOllamaModels() {
    let baseUrl = ollamaBaseUrlInput.value.trim();
    if (baseUrl.includes(".cloudshell.dev")) {
        baseUrl = "";
        ollamaBaseUrlInput.value = "";
        addLog("INFO", "OLLAMA", "Cleared Cloud Shell preview URL; using backend local OLLAMA_BASE_URL.");
    }

    discoverOllamaBtn.disabled = true;
    discoverOllamaBtn.textContent = "Scanning...";
    showStatusBanner(
        ollamaDiscoveryStatus,
        `Contacting Ollama (${baseUrl || "backend local default"})...`,
        "info"
    );
    addLog("INFO", "OLLAMA", `Starting dynamic model discovery (${baseUrl || "backend default"})`);

    try {
        const endpoint = baseUrl
            ? `/api/ollama/models?base_url=${encodeURIComponent(baseUrl)}`
            : "/api/ollama/models";
        const response = await fetch(endpoint);
        const data = await response.json();

        if (!response.ok) {
            const detail = data.detail || `HTTP ${response.status}`;
            throw new Error(detail);
        }

        ollamaModelSelect.innerHTML = "";
        if (!data.models || data.models.length === 0) {
            const emptyOpt = document.createElement("option");
            emptyOpt.value = "";
            emptyOpt.textContent = "-- No local models found in Ollama --";
            ollamaModelSelect.appendChild(emptyOpt);
            showStatusBanner(
                ollamaDiscoveryStatus,
                `Connected to Ollama at ${data.base_url}, but 0 models were returned. Pull a model first.`,
                "info"
            );
            addLog("WARN", "OLLAMA", "Connected to Ollama, 0 models installed.", data.base_url);
            return;
        }

        for (const model of data.models) {
            const opt = document.createElement("option");
            opt.value = model.name;
            const sizeLabel = formatBytes(model.size);
            opt.textContent = sizeLabel ? `${model.name} (${sizeLabel})` : model.name;
            ollamaModelSelect.appendChild(opt);
        }

        const preferred = ollamaCustomModelInput.value.trim() || data.models[0].name;
        const hasPreferred = data.models.some((m) => m.name === preferred);
        ollamaModelSelect.value = hasPreferred ? preferred : data.models[0].name;
        ollamaCustomModelInput.value = ollamaModelSelect.value;

        showStatusBanner(
            ollamaDiscoveryStatus,
            `Discovered ${data.count} Ollama model(s) at ${data.base_url}.`,
            "success"
        );
        addLog(
            "INFO",
            "OLLAMA",
            `Discovered ${data.count} model(s).`,
            data.models.map((m) => m.name).join(", ")
        );
    } catch (err) {
        showStatusBanner(
            ollamaDiscoveryStatus,
            `Discovery failed: ${err.message}. Open Logs for troubleshooting details.`,
            "error"
        );
        addLog("ERROR", "OLLAMA", "Model discovery failed.", err.message);
    } finally {
        discoverOllamaBtn.disabled = false;
        discoverOllamaBtn.textContent = "Discover Models";
    }
}

function getFileExtension(filename) {
    const idx = filename.lastIndexOf(".");
    return idx >= 0 ? filename.slice(idx).toLowerCase() : "";
}

async function handleAttachmentSelected(file) {
    if (!file || isUploading || isSending) {
        return;
    }

    const ext = getFileExtension(file.name);
    const isImage = IMAGE_EXTENSIONS.includes(ext) || file.type.startsWith("image/");
    const isDoc = DOC_EXTENSIONS.includes(ext);

    if (!isImage && !isDoc) {
        const msg = `Unsupported file "${file.name}". Allowed: JPG, JPEG, PNG, WEBP, PDF, TXT.`;
        appendMessage("assistant", msg, "error");
        addLog("ERROR", "UPLOAD", msg);
        attachmentFileInput.value = "";
        return;
    }

    const endpoint = isDoc ? "/upload/document" : "/upload/image";
    isUploading = true;
    attachFileBtn.disabled = true;
    updateSendButtonState();

    addLog("INFO", "UPLOAD", `Uploading ${isDoc ? "document" : "image"}...`, {
        filename: file.name,
        size: file.size,
        endpoint,
    });

    try {
        const formData = new FormData();
        formData.append("file", file);

        const response = await fetch(endpoint, {
            method: "POST",
            body: formData,
        });

        let data = null;
        try {
            data = await response.json();
        } catch (parseErr) {
            throw new Error(`Invalid JSON response (HTTP ${response.status})`);
        }

        if (!response.ok) {
            throw new Error(extractApiErrorMessage(data, response.status));
        }

        if (isDoc) {
            showPendingDocument(data);
            addLog("INFO", "UPLOAD", "Document uploaded & text extracted.", {
                fileId: data.file_id,
                fileType: data.file_type,
                pages: data.page_count,
                chars: data.char_count,
                truncated: data.truncated,
            });
        } else {
            showPendingImage(data);
            addLog("INFO", "UPLOAD", "Image uploaded and stored.", {
                fileId: data.file_id,
                url: data.url,
                dimensions: `${data.width}x${data.height}`,
                sizeBytes: data.size_bytes,
            });
        }
    } catch (err) {
        clearPendingAttachment();
        appendMessage("assistant", `Upload failed: ${err.message}`, "error");
        addLog("ERROR", "UPLOAD", "File upload failed.", err.message);
    } finally {
        isUploading = false;
        attachFileBtn.disabled = false;
        updateSendButtonState();
    }
}

function renderOutputIntoBubble(bubble, output) {
    bubble.innerHTML = "";
    const text = output && output.text ? output.text : "";
    if (text) {
        const textNode = document.createElement("div");
        textNode.textContent = text;
        bubble.appendChild(textNode);
    }

    if (
        output &&
        output.used_a2ui &&
        Array.isArray(output.a2ui_messages) &&
        output.a2ui_messages.length > 0 &&
        window.OracleA2uiRenderer
    ) {
        const surfaceRoot = window.OracleA2uiRenderer.renderA2uiMessages(
            output.a2ui_messages,
            handleA2uiAction
        );
        bubble.appendChild(surfaceRoot);
    }
}

function appendMessage(role, text, extraClass = "", attachment = null, output = null) {
    const emptyEl = document.getElementById("empty-state");
    if (emptyEl && emptyEl.parentNode) {
        emptyEl.remove();
    }

    const row = document.createElement("div");
    row.className = `message-row ${role}`;

    const roleLabel = document.createElement("span");
    roleLabel.className = "message-role";
    roleLabel.textContent = role === "user" ? "You" : "Assistant";

    const bubble = document.createElement("div");
    bubble.className = extraClass ? `message-bubble ${extraClass}` : "message-bubble";

    if (attachment && attachment.kind === "image" && attachment.data.url) {
        const img = document.createElement("img");
        img.className = "chat-image-preview";
        img.src = attachment.data.url;
        img.alt = attachment.data.original_filename || "Uploaded image";
        bubble.appendChild(img);

        const caption = document.createElement("span");
        caption.className = "chat-image-caption";
        caption.textContent = `${attachment.data.original_filename} (${attachment.data.width}x${attachment.data.height})`;
        bubble.appendChild(caption);
    } else if (attachment && attachment.kind === "document") {
        const pill = document.createElement("div");
        pill.className = "chat-doc-pill";

        const tag = document.createElement("span");
        tag.className = "chat-doc-tag";
        tag.textContent = attachment.data.file_type || "DOC";

        const name = document.createElement("span");
        name.textContent = `${attachment.data.original_filename} (${attachment.data.page_count}p · ${attachment.data.char_count.toLocaleString()} chars)`;

        pill.appendChild(tag);
        pill.appendChild(name);
        bubble.appendChild(pill);
    }

    if (output) {
        renderOutputIntoBubble(bubble, output);
    } else if (text) {
        const textNode = document.createElement("div");
        textNode.textContent = text;
        bubble.appendChild(textNode);
    }

    row.appendChild(roleLabel);
    row.appendChild(bubble);
    chatHistory.appendChild(row);
    scrollToBottom();

    return bubble;
}

function extractApiErrorMessage(data, statusCode) {
    if (!data) {
        return `HTTP ${statusCode}`;
    }
    if (typeof data.detail === "string") {
        return data.detail;
    }
    if (Array.isArray(data.detail) && data.detail.length > 0) {
        return data.detail.map((item) => item.msg || JSON.stringify(item)).join("; ");
    }
    return `HTTP ${statusCode}`;
}

        });
    } catch (err) {

    }
}

async function handleA2uiAction(actionPayload) {
    addLog("INFO", "A2UI_ACTION", `Triggered action '${actionPayload.name}'`, actionPayload);
    if (isSending || isUploading) {
        return;
    }

    const actionMessage = `UI Action "${actionPayload.name}" submitted with values: ${JSON.stringify(actionPayload.context)}`;
    await sendChatRequest({
        promptText: actionMessage,
        a2uiAction: actionPayload,
        userDisplayText: `[A2UI Action: ${actionPayload.name}] ${JSON.stringify(actionPayload.context)}`,
    });
}

async function sendChatRequest({
    promptText,
    attached = null,
    a2uiAction = null,
    userDisplayText = null,
}) {
    const isVision = Boolean(attached && attached.kind === "image");
    const isDocument = Boolean(attached && attached.kind === "document");
    const uiMode = uiModeSelect.value || "auto";
    const activeProvider = currentSettings.provider || "openrouter";

    appendMessage("user", userDisplayText || promptText, "", attached);

    isSending = true;
    updateSendButtonState();

    const routerModel =
        activeProvider === "ollama"
            ? currentSettings.ollamaModel
            : currentSettings.openrouterModel || DEFAULT_SETTINGS.openrouterModel;

    const activePin =
        activeProvider === "openrouter" && routerModel === "openrouter/free"
            ? isVision
                ? currentSettings.pinnedVisionModel
                : currentSettings.pinnedOpenRouterModel
            : "";

    const effectiveDisplay = activePin || routerModel || activeProvider;

    const statusText = isVision
        ? `Analyzing image (${effectiveDisplay})...`
        : isDocument
          ? `Reading document (${effectiveDisplay})...`
          : `Thinking (${effectiveDisplay})...`;
    const pendingBubble = appendMessage("assistant", statusText, "pending");

    const requestBody = {
        message: promptText,
        session_id: currentSessionId,
        provider: activeProvider,
        model: routerModel,
        ui_mode: uiMode,
    };
    if (isVision) {
        requestBody.image_id = attached.data.file_id;
    }
    if (isDocument) {
        requestBody.document_id = attached.data.file_id;
        requestBody.document_name = attached.data.original_filename;
    }
    if (a2uiAction) {
        requestBody.a2ui_action = a2uiAction;
    }
    if (activeProvider === "openrouter") {
        if (routerModel === "openrouter/free") {
            if (isVision && currentSettings.pinnedVisionModel) {
                requestBody.pinned_vision_model = currentSettings.pinnedVisionModel;
            } else if (!isVision && currentSettings.pinnedOpenRouterModel) {
                requestBody.pinned_model = currentSettings.pinnedOpenRouterModel;
            }
        }
        if (currentSettings.openrouterApiKey) {
            requestBody.api_key = currentSettings.openrouterApiKey;
        }
    } else if (activeProvider === "ollama" && currentSettings.ollamaBaseUrl) {
        requestBody.ollama_base_url = currentSettings.ollamaBaseUrl;
    }

    const logSource = isVision ? "VISION" : isDocument ? "DOCUMENT" : "CHAT";
    addLog("INFO", logSource, "POST /chat request sent.", {
        sessionId: currentSessionId,
        provider: activeProvider,
        model: routerModel || null,
        uiMode,
        vision: isVision,
        document: isDocument,
        hasAction: Boolean(a2uiAction),
        pinnedModel: activePin || null,
    });

    try {
        const response = await fetch("/chat", {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
            },
            body: JSON.stringify(requestBody),
        });

        let data = null;
        try {
            data = await response.json();
        } catch (parseErr) {
            throw new Error(`Invalid JSON response (HTTP ${response.status})`);
        }

        if (!response.ok) {
            throw new Error(extractApiErrorMessage(data, response.status));
        }

        if (data.session_id) {
            currentSessionId = data.session_id;
            localStorage.setItem(SESSION_STORAGE_KEY, currentSessionId);
        }

        if (
            activeProvider === "openrouter" &&
            routerModel === "openrouter/free" &&
            data.model &&
            data.model !== "openrouter/free"
        ) {
            if (data.vision_used) {
                if (currentSettings.pinnedVisionModel !== data.model) {
                    currentSettings.pinnedVisionModel = data.model;
                    saveSettingsToStorage(currentSettings);
                    pinnedVisionModelInput.value = data.model;
                    addLog("INFO", "OPENROUTER", `Assigned & pinned free vision model: ${data.model}`);
                }
            } else if (currentSettings.pinnedOpenRouterModel !== data.model) {
                currentSettings.pinnedOpenRouterModel = data.model;
                saveSettingsToStorage(currentSettings);
                pinnedFreeModelInput.value = data.model;
                updateHeaderBadge();
                addLog("INFO", "OPENROUTER", `Assigned & pinned free model: ${data.model}`);
            }
        }

        pendingBubble.classList.remove("pending");
        const standardized = data.output || { text: data.reply ?? "", used_a2ui: false, a2ui_messages: [] };
        renderOutputIntoBubble(pendingBubble, standardized);

        if (standardized.repair_retried && standardized.used_a2ui) {
            addLog("INFO", "A2UI", "Auto-repaired A2UI syntax on 1-retry.");
        }
        if (Array.isArray(standardized.validation_warnings) && standardized.validation_warnings.length > 0) {
            addLog("WARN", "A2UI", "Fallback to plain text after A2UI validation warning.", standardized.validation_warnings);
        }

        addLog("INFO", logSource, "Received standardized reply.", {
            sessionId: data.session_id,
            historyLength: data.history_length,
            provider: data.provider || activeProvider,
            model: data.model || effectiveDisplay,
            usedA2ui: Boolean(standardized.used_a2ui),
            repairRetried: Boolean(standardized.repair_retried),
            rawFormat: standardized.raw_format,
            a2uiMessages: (standardized.a2ui_messages || []).length,
        });
    } catch (err) {
        pendingBubble.classList.remove("pending");
        pendingBubble.classList.add("error");
        pendingBubble.textContent = `Error: ${err.message}`;
        addLog("ERROR", logSource, `${activeProvider} request failed.`, err.message);
    } finally {
        isSending = false;
        updateSendButtonState();
        messageInput.focus();
        scrollToBottom();
    }
}

async function handleSend(event) {
    event.preventDefault();

    const rawText = messageInput.value.trim();
    const attached = pendingAttachment;

    if ((!rawText && !attached) || isSending || isUploading) {
        return;
    }

    const isVision = Boolean(attached && attached.kind === "image");
    let promptText = rawText;
    if (!promptText) {
        promptText = isVision
            ? "Describe this image in detail."
            : "Summarize this document and list its key points.";
    }

    messageInput.value = "";
    autoResizeInput();
    if (attached) {
        clearPendingAttachment();
    }

    await sendChatRequest({ promptText, attached });
}

class TestProviderAbstraction(unittest.TestCase):
    def setUp(self) -> None:
        self.client = TestClient(app)

    def test_factory_creates_openrouter_and_ollama_providers(self) -> None:
        settings = get_settings()
        name_or, prov_or = create_llm_provider(settings, provider_name="openrouter", api_key="k")
        self.assertEqual(name_or, "openrouter")
        self.assertIsInstance(prov_or, OpenRouterProvider)
        self.assertIsInstance(prov_or, LLMProvider)

        name_ol, prov_ol = create_llm_provider(settings, provider_name="ollama", default_model="gemma3:4b")
        self.assertEqual(name_ol, "ollama")
        self.assertIsInstance(prov_ol, OllamaProvider)
        self.assertIsInstance(prov_ol, LLMProvider)

    def test_factory_rejects_unknown_provider(self) -> None:
        settings = get_settings()
        with self.assertRaises(LLMProviderError) as ctx:
            create_llm_provider(settings, provider_name="unknown_llm")
        self.assertEqual(ctx.exception.status_code, 400)

    def test_chat_routes_via_ollama_when_configured_by_env(self) -> None:
        mock_reply = LLMResponse(content="Hello from local Ollama!", model="gemma3:4b")
        with patch.dict(os.environ, {"LLM_PROVIDER": "ollama", "MODEL_NAME": "gemma3:4b"}):
            with patch("backend.llm.ollama.OllamaProvider.chat", new=AsyncMock(return_value=mock_reply)):
                resp = self.client.post(

                    json={"message": "Hi", "ui_mode": "text"},
                )

        self.assertEqual(resp.status_code, 200)
        body = resp.json()
        self.assertEqual(body["provider"], "ollama")
        self.assertEqual(body["model"], "gemma3:4b")
        self.assertEqual(body["reply"], "Hello from local Ollama!")

    def test_chat_routes_via_openrouter_when_configured_by_env(self) -> None:
        mock_reply = LLMResponse(content="Hello from OpenRouter!", model="test/model:free")
        with patch.dict(os.environ, {"LLM_PROVIDER": "openrouter", "OPENROUTER_API_KEY": "test-key"}):

    def test_ollama_analyze_image_strips_data_url_prefix(self) -> None:

        self.assertEqual(captured["messages"][-1]["images"], ["QUJDREVGRw=="])
